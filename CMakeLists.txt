cmake_minimum_required(VERSION 3.20)
project(maimunas_queue LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(queuecore src/simulator.cpp)
target_sources(queuecore PRIVATE src/experiment.cpp src/config_validation.cpp)
target_include_directories(queuecore PUBLIC include)
target_compile_options(queuecore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(queue-sim src/main.cpp)
target_link_libraries(queue-sim PRIVATE queuecore)
enable_testing()
add_executable(queue-tests tests/tests.cpp)
target_sources(queue-tests PRIVATE tests/experiment_tests.cpp)
target_link_libraries(queue-tests PRIVATE queuecore)
add_test(NAME queue-tests COMMAND queue-tests)
